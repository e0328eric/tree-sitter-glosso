wrapping :: (value: s64, left: s64, right: s64) -> s64 {
  negative := -% value;
  negative_cast := -% cast(s64)value;
  cast_negative := cast(s64)-%value;
  difference := left -% right;
  return negative + negative_cast + cast_negative + difference;
}
