  #memory{returns_borrow(data), reads(data), noescape(data)};

#memory Runtime.acquire {
  returns_fresh,
  released_by(place: returned, by: Runtime.destroy),
  noescape(arg(0)),
};

main :: () {
  value := Runtime.acquire()
    #memory{released_by(Runtime.destroy), noescape(arg(0))};
  #memory trusted("external allocator invariant") {
    Runtime.use(value);
  }
}
