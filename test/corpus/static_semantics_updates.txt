  #assert true, "ready";
  #assert #comptime 1 == 1;
  if true run()?;
  prefix := !! 7;
  suffix := prefix ++;

  if #pattern {
  case true; run();
  else; return;
  }

  item := Left(5);
  if #pattern Left(value) | Right(value) = item run();
  pair := .{ .first = 5, .second = 9 };
  if #pattern whole @ .{ .first = 0..=9, ... } = pair run();
  values: [3]s64 = .[1, 2, 3];
  if #pattern .[head, ..., tail] = values[..] run();

  label := ifx true "yes" else "no";
  grade := ifx 90 == {
  case 90; "A";
  else; "B";
  };
  number := ifx #pattern item == {
  case Left(value) if value > 0; value;
  else; 0;
  };
}
